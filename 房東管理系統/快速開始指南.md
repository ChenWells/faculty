# 房東管理系統快速開始指南

## 🎯 專案目標

建立一個**完整的房東管理系統**，電費計算作為租金計算的輔助功能。

## 📋 核心功能優先級

### 第一階段：核心業務 (2週)
1. **房屋管理** - 房屋資料、狀態追蹤
2. **租客管理** - 租客資料、聯絡資訊
3. **合約管理** - 合約建立、狀態管理
4. **收租管理** - 月度租金計算、繳費記錄

### 第二階段：電費整合 (1週)
1. **電費計算** - 三種計算方式
2. **費用整合** - 與收租管理整合

### 第三階段：通知同步 (1週)
1. **LINE通知** - 收租通知、繳費確認
2. **Google Sheets** - 資料同步

'use client';

import { useState, useEffect } from 'react';
import axios from 'axios';

interface Property {
  id: string;
  name: string;
  address: string;
  monthly_rent: number;
  status: string;
}

export default function PropertyList() {
  const [properties, setProperties] = useState<Property[]>([]);
  const [loading, setLoading] = useState(true);

  useEffect(() => {
    fetchProperties();
  }, []);

  const fetchProperties = async () => {
    try {
      const response = await axios.get('http://localhost:3001/api/properties');
      setProperties(response.data);
    } catch (error) {
      console.error('取得房屋資料失敗:', error);
    } finally {
      setLoading(false);
    }
  };

  if (loading) {
    return <div className="text-center py-8">載入中...</div>;
  }

  return (
    <div className="bg-white shadow rounded-lg p-6">
      <h2 className="text-xl font-bold mb-4">房屋管理</h2>
      
      <div className="grid grid-cols-1 md:grid-cols-2 lg:grid-cols-3 gap-4">
        {properties.map((property) => (
          <div key={property.id} className="border rounded-lg p-4">
            <h3 className="font-semibold">{property.name}</h3>
            <p className="text-gray-600 text-sm">{property.address}</p>
            <p className="text-gray-800">月租金: ${property.monthly_rent}</p>
            <span className={`inline-block px-2 py-1 text-xs rounded ${
              property.status === 'available' ? 'bg-green-100 text-green-800' :
              property.status === 'rented' ? 'bg-blue-100 text-blue-800' :

               property.status === 'rented' ? '已出租' : '維修中'}

  );
}

### 4. 電費計算整合 (1週後)

#### 4.1 電費計算服務

```javascript
// backend/src/services/ElectricityService.js
class ElectricityService {
  // 計算電費
  async calculateElectricity(contractId, totalUsage, method = 'average_rate') {
    const contract = await Contract.findByPk(contractId);
    let rate = 0;

    switch (method) {
      case 'fixed_rate':
        rate = contract.fixed_electricity_rate || 3.5;
        break;
      case 'average_rate':
        rate = await this.getAverageRate(contract.property_id);
        break;
      case 'mixed_rate':
        rate = await this.getMixedRate(contract.property_id);
        break;
    }

    return {
      total_usage: totalUsage,
      total_amount: totalUsage * rate,
      rate: rate
    };
  }

  // 與收租記錄整合
  async integrateWithRental(contractId, billingMonth, electricityData) {
    const rentalRecord = await RentalRecord.findOne({
      where: { contract_id: contractId, billing_month: billingMonth }
    });

    if (rentalRecord) {
      rentalRecord.electricity_fee = electricityData.total_amount;
      rentalRecord.total_amount = rentalRecord.monthly_rent + electricityData.total_amount;
      await rentalRecord.save();
    }

    return rentalRecord;
  }
}
```

#### 4.2 電費計算組件

```typescript
// frontend/src/components/ElectricityCalculator.tsx
'use client';

import { useState } from 'react';
import axios from 'axios';

interface ElectricityCalculatorProps {
  contractId: string;
  onCalculate: (result: any) => void;
}

export default function ElectricityCalculator({ contractId, onCalculate }: ElectricityCalculatorProps) {
  const [formData, setFormData] = useState({
    total_usage: 0,
    calculation_method: 'average_rate'
  });
  
  const [result, setResult] = useState(null);
  const [loading, setLoading] = useState(false);

  const handleCalculate = async (e: React.FormEvent) => {
    e.preventDefault();
    setLoading(true);
    
    try {
      const response = await axios.post('http://localhost:3001/api/electricity/calculate', {
        ...formData,
        contract_id: contractId
      });
      
      setResult(response.data);
      onCalculate(response.data);
    } catch (error) {
      console.error('計算失敗:', error);
    } finally {
      setLoading(false);
    }
  };

  return (
    <div className="bg-white shadow rounded-lg p-6">
      <h3 className="text-lg font-medium mb-4">電費計算</h3>
      
      <form onSubmit={handleCalculate} className="space-y-4">
        <div>
          <label className="block text-sm font-medium text-gray-700">
            總用電度數
          </label>
          <input
            type="number"
            step="0.01"
            value={formData.total_usage}
            onChange={(e) => setFormData({
              ...formData,
              total_usage: parseFloat(e.target.value)
            })}
            className="mt-1 block w-full rounded-md border-gray-300 shadow-sm"
            required
          />
        </div>
        
        <div>
          <label className="block text-sm font-medium text-gray-700">
            計算方式
          </label>
          <select
            value={formData.calculation_method}
            onChange={(e) => setFormData({
              ...formData,
              calculation_method: e.target.value
            })}
            className="mt-1 block w-full rounded-md border-gray-300 shadow-sm"
          >
            <option value="fixed_rate">固定電價</option>
            <option value="average_rate">平均電價</option>
            <option value="mixed_rate">混合電價</option>
          </select>
        </div>
        
        <button
          type="submit"
          disabled={loading}
          className="w-full bg-blue-600 text-white py-2 px-4 rounded-md hover:bg-blue-700 disabled:opacity-50"
        >
          {loading ? '計算中...' : '計算電費'}
        </button>
      </form>
      
      {result && (
        <div className="mt-6 bg-blue-50 p-4 rounded">
          <h4 className="font-medium text-blue-900">計算結果</h4>
          <div className="mt-2 space-y-2">
            <p className="text-sm text-blue-800">
              電費總額: <span className="font-medium">${result.total_amount.toFixed(2)}</span>
            </p>
            <p className="text-sm text-blue-800">
              每度電費: <span className="font-medium">${result.rate.toFixed(4)}</span>
            </p>
          </div>
        </div>
      )}
    </div>
  );
}
```
