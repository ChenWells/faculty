# 電費管理系統實作指南

## 🎯 專案目標

基於您的Google試算表，建立一個專門的電費管理Web應用，實現：
- 電費資料的數位化管理
- 自動電費計算和分攤
- 與Google試算表的即時同步
- 電費趨勢分析和報表

## 📊 試算表分析

### 資料結構
- **時間範圍**：113年11~12月 到 114年11~12月
- **房屋單位**：自強1樓、自強2樓、自強3樓、自強4~5樓、中央MBA
- **核心欄位**：度數、電費、房客度數加總、房客電費加總
- **計算邏輯**：根據前期平均電費計算每度電費

### 第五步：Google Sheets同步服務

```javascript
// src/services/GoogleSheetsService.js
const { google } = require('googleapis');
const fs = require('fs');

class GoogleSheetsService {
  constructor() {
    const keyFile = JSON.parse(
      fs.readFileSync(process.env.GOOGLE_SERVICE_ACCOUNT_KEY_PATH)
    );
    
    this.auth = new google.auth.GoogleAuth({
      credentials: keyFile,
      scopes: ['https://www.googleapis.com/auth/spreadsheets']
    });
    
    this.sheets = google.sheets({ version: 'v4', auth: this.auth });
    this.spreadsheetId = process.env.GOOGLE_SHEETS_ID;
  }

  // 讀取試算表資料
  async readSheetData() {
    try {
      const response = await this.sheets.spreadsheets.values.get({
        spreadsheetId: this.spreadsheetId,
        range: 'A1:Z100'
      });

      return this.parseSheetData(response.data.values);
    } catch (error) {
      console.error('讀取Google Sheets失敗:', error);
      throw error;
    }
  }

  // 解析試算表資料
  parseSheetData(values) {
    const records = [];
    const properties = ['自強1樓', '自強2樓', '自強3樓', '自強4~5樓', '中央MBA'];
    
    // 解析每一行資料
    for (let i = 3; i < values.length; i++) {
      const row = values[i];
      if (!row[0] || !properties.includes(row[0])) continue;
      
      // 解析每個時期的資料
      for (let j = 1; j < row.length; j += 4) {
        if (row[j] && row[j + 1]) {
          const period = this.getPeriodFromColumn(j);
          if (period) {
            records.push({
              property_name: row[0],
              period_start: period.start,
              period_end: period.end,
              total_usage: parseFloat(row[j]) || 0,
              total_amount: parseFloat(row[j + 1]) || 0,
              tenant_usage: parseFloat(row[j + 2]) || 0,
              tenant_amount: parseFloat(row[j + 3]) || 0
            });
          }
        }
      }
    }
    
    return records;
  }

  // 根據欄位位置取得期間
  getPeriodFromColumn(columnIndex) {
    const periods = [
      { start: '2023-11-01', end: '2023-12-31' },
      { start: '2024-01-01', end: '2024-02-29' },
      { start: '2024-03-01', end: '2024-04-30' },
      { start: '2024-05-01', end: '2024-06-30' },
      { start: '2024-07-01', end: '2024-08-31' },
      { start: '2024-09-01', end: '2024-10-31' },
      { start: '2024-11-01', end: '2024-12-31' }
    ];
    
    const periodIndex = Math.floor((columnIndex - 1) / 4);
    return periods[periodIndex];
  }

  // 更新試算表
  async updateSheet(data) {
    try {
      const values = this.formatDataForSheet(data);
      
      await this.sheets.spreadsheets.values.update({
        spreadsheetId: this.spreadsheetId,
        range: 'A4:Z100',
        valueInputOption: 'RAW',
        resource: { values }
      });
      
      console.log('Google Sheets更新成功');
    } catch (error) {
      console.error('更新Google Sheets失敗:', error);
      throw error;
    }
  }

  // 格式化資料為試算表格式
  formatDataForSheet(data) {
    // 將資料轉換為試算表格式
    return data.map(record => [
      record.property_name,
      record.total_usage,
      record.total_amount,
      record.tenant_usage,
      record.tenant_amount
    ]);
  }
}

module.exports = GoogleSheetsService;
```

### 第六步：電費計算服務

```javascript
// src/services/ElectricityCalculationService.js
const ElectricityRecord = require('../models/ElectricityRecord');
const Property = require('../models/Property');

class ElectricityCalculationService {
  // 計算平均電價
  async calculateAverageRate(propertyId, months = 6) {
    try {
      const records = await ElectricityRecord.findAll({
        where: { property_id: propertyId },
        order: [['period_start', 'DESC']],
        limit: months
      });

      if (records.length === 0) {
        return 0;
      }

      const totalAmount = records.reduce((sum, record) => sum + parseFloat(record.total_amount), 0);
      const totalUsage = records.reduce((sum, record) => sum + parseFloat(record.total_usage), 0);

      return totalUsage > 0 ? totalAmount / totalUsage : 0;
    } catch (error) {
      console.error('計算平均電價失敗:', error);
      throw error;
    }
  }

  // 計算電費
  async calculateElectricityBill(propertyId, totalUsage, calculationMethod = 'average_rate', fixedRate = 5.0) {
    try {
      let rate = 0;

      switch (calculationMethod) {
        case 'fixed_rate':
          rate = fixedRate;
          break;
        case 'average_rate':
          rate = await this.calculateAverageRate(propertyId);
          break;
        case 'mixed_rate':
          // 混合計算：私電用平均電價，公電用固定電價
          const averageRate = await this.calculateAverageRate(propertyId);
          // 這裡需要根據實際業務邏輯調整
          rate = averageRate;
          break;
        default:
          rate = await this.calculateAverageRate(propertyId);
      }

      const totalAmount = totalUsage * rate;
      
      return {
        total_usage: totalUsage,
        total_amount: totalAmount,
        average_rate: rate,
        calculation_method: calculationMethod
      };
    } catch (error) {
      console.error('計算電費失敗:', error);
      throw error;
    }
  }

  // 計算房客分攤
  calculateTenantSplit(totalAmount, tenantUsage, totalUsage) {
    if (totalUsage === 0) return 0;
    return (tenantUsage / totalUsage) * totalAmount;
  }
}

module.exports = ElectricityCalculationService;
```

### 第七步：API路由

```javascript
// src/routes/electricity.js
const express = require('express');
const router = express.Router();
const ElectricityRecord = require('../models/ElectricityRecord');
const Property = require('../models/Property');
const GoogleSheetsService = require('../services/GoogleSheetsService');
const ElectricityCalculationService = require('../services/ElectricityCalculationService');

const sheetsService = new GoogleSheetsService();
const calculationService = new ElectricityCalculationService();

// 取得所有電費記錄
router.get('/', async (req, res) => {
  try {
    const records = await ElectricityRecord.findAll({
      include: [{ model: Property, as: 'property' }],
      order: [['period_start', 'DESC']]
    });
    
    res.json(records);
  } catch (error) {
    res.status(500).json({ error: error.message });
  }
});

// 新增電費記錄
router.post('/', async (req, res) => {
  try {
    const record = await ElectricityRecord.create(req.body);
    res.status(201).json(record);
  } catch (error) {
    res.status(400).json({ error: error.message });
  }
});

// 計算電費
router.post('/calculate', async (req, res) => {
  try {
    const { property_id, total_usage, calculation_method, fixed_rate } = req.body;
    
    const result = await calculationService.calculateElectricityBill(
      property_id,
      total_usage,
      calculation_method,
      fixed_rate
    );
    
    res.json(result);
  } catch (error) {
    res.status(400).json({ error: error.message });
  }
});

// 同步Google Sheets資料
router.post('/sync', async (req, res) => {
  try {
    const sheetData = await sheetsService.readSheetData();
    
    // 更新資料庫
    for (const data of sheetData) {
      const property = await Property.findOne({ where: { name: data.property_name } });
      if (property) {
        await ElectricityRecord.upsert({
          property_id: property.id,
          period_start: data.period_start,
          period_end: data.period_end,
          total_usage: data.total_usage,
          total_amount: data.total_amount,
          tenant_usage: data.tenant_usage,
          tenant_amount: data.tenant_amount
        });
      }
    }
    
    res.json({ message: '同步成功', count: sheetData.length });
  } catch (error) {
    res.status(500).json({ error: error.message });
  }
});

module.exports = router;
```

```typescript
// src/components/ElectricityCalculator.tsx
'use client';

import { useState } from 'react';
import axios from 'axios';

interface CalculationResult {
  total_usage: number;
  total_amount: number;
  average_rate: number;
  calculation_method: string;
}

export default function ElectricityCalculator() {
  const [formData, setFormData] = useState({
    property_id: '',
    total_usage: 0,
    calculation_method: 'average_rate',
    fixed_rate: 5.0
  });
  
  const [result, setResult] = useState<CalculationResult | null>(null);
  const [loading, setLoading] = useState(false);

  const handleCalculate = async (e: React.FormEvent) => {
    e.preventDefault();
    setLoading(true);
    
    try {
      const response = await axios.post('/api/electricity/calculate', formData);
      setResult(response.data);
    } catch (error) {
      console.error('計算失敗:', error);
    } finally {
      setLoading(false);
    }
  };

  return (
    <div className="bg-white shadow rounded-lg">
      <div className="px-4 py-5 sm:p-6">
        <h3 className="text-lg leading-6 font-medium text-gray-900 mb-4">
          電費計算器
        </h3>
        
        <form onSubmit={handleCalculate} className="space-y-4">
          <div>
            <label className="block text-sm font-medium text-gray-700">
              總用電度數
            </label>
            <input
              type="number"
              step="0.01"
              value={formData.total_usage}
              onChange={(e) => setFormData({
                ...formData,
                total_usage: parseFloat(e.target.value)
              })}
              className="mt-1 block w-full rounded-md border-gray-300 shadow-sm"
              required
            />
          </div>
          
          <div>
            <label className="block text-sm font-medium text-gray-700">
              計算方式
            </label>
            <select
              value={formData.calculation_method}
              onChange={(e) => setFormData({
                ...formData,
                calculation_method: e.target.value
              })}
              className="mt-1 block w-full rounded-md border-gray-300 shadow-sm"
            >
              <option value="fixed_rate">固定電價</option>
              <option value="average_rate">平均電價</option>
              <option value="mixed_rate">混合電價</option>
            </select>
          </div>
          
          {formData.calculation_method === 'fixed_rate' && (
            <div>
              <label className="block text-sm font-medium text-gray-700">
                固定電價 (元/度)
              </label>
              <input
                type="number"
                step="0.01"
                value={formData.fixed_rate}
                onChange={(e) => setFormData({
                  ...formData,
                  fixed_rate: parseFloat(e.target.value)
                })}
                className="mt-1 block w-full rounded-md border-gray-300 shadow-sm"
              />
            </div>
          )}
          
          <button
            type="submit"
            disabled={loading}
            className="w-full bg-blue-600 text-white py-2 px-4 rounded-md hover:bg-blue-700 disabled:opacity-50"
          >
            {loading ? '計算中...' : '計算電費'}
          </button>
        </form>
        
        {result && (

            <h4 className="font-medium text-blue-900">計算結果</h4>

              </p>
            </div>
          </div>
        )}
      </div>

  );
```
