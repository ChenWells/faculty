```javascript
// src/services/ElectricityCalculationService.js
class ElectricityCalculationService {
  // 計算電費
  async calculateElectricityBill(contractId, totalUsage, calculationMethod = 'average_rate') {
    try {
      const contract = await Contract.findByPk(contractId);
      let rate = 0;

      switch (calculationMethod) {
        case 'fixed_rate':
          rate = contract.fixed_electricity_rate;
          break;
        case 'average_rate':
          rate = await this.calculateAverageRate(contract.property_id);
          break;
        case 'mixed_rate':
          rate = await this.calculateMixedRate(contract.property_id, contract.fixed_electricity_rate);
          break;
      }

      const totalAmount = totalUsage * rate;
      
      return {
        total_usage: totalUsage,
        total_amount: totalAmount,
        electricity_rate: rate,
        calculation_method: calculationMethod
      };
    } catch (error) {
      throw error;
    }
  }

  // 與收租記錄整合
  async integrateWithRentalRecord(contractId, billingMonth, electricityData) {
    try {
      const rentalRecord = await RentalRecord.findOne({
        where: { contract_id: contractId, billing_month: billingMonth }
      });

      if (rentalRecord) {
        rentalRecord.electricity_fee = electricityData.total_amount;
        rentalRecord.total_amount = rentalRecord.monthly_rent + electricityData.total_amount;
        await rentalRecord.save();
      }

      return rentalRecord;
    } catch (error) {
      throw error;
    }
  }
}
```

```typescript
'use client';

import { useState } from 'react';
import axios from 'axios';

interface ElectricityCalculatorProps {
  contractId: string;
  onCalculate: (result: any) => void;
}

export default function ElectricityCalculator({ contractId, onCalculate }: ElectricityCalculatorProps) {
  const [formData, setFormData] = useState({
    total_usage: 0,
    calculation_method: 'average_rate'
  });
  
  const [result, setResult] = useState(null);
  const [loading, setLoading] = useState(false);

  const handleCalculate = async (e: React.FormEvent) => {
    e.preventDefault();
    setLoading(true);
    
    try {
      const response = await axios.post('/api/electricity/calculate', {
        ...formData,
        contract_id: contractId
      });
      
      setResult(response.data);
      onCalculate(response.data);
    } catch (error) {
      console.error('計算失敗:', error);
    } finally {
      setLoading(false);
    }
  };

  return (

            總用電度數
          </label>
          <input
            type="number"
            step="0.01"
            value={formData.total_usage}
            onChange={(e) => setFormData({
              ...formData,
              total_usage: parseFloat(e.target.value)
            })}
```

```javascript
// src/services/RentalCalculationService.js
class RentalCalculationService {
  // 自動計算月度租金
  async calculateMonthlyRent(contractId, billingMonth) {
    try {
      const contract = await Contract.findByPk(contractId, {
        include: [{ model: Property, as: 'property' }]
      });

      // 基礎租金
      const monthlyRent = contract.monthly_rent;
      
      // 電費計算
      const electricityFee = await this.calculateElectricityFee(contractId, billingMonth);
      
      // 其他費用
      const otherFees = contract.management_fee + contract.cleaning_fee + contract.internet_fee;
      
      // 總金額
      const totalAmount = monthlyRent + electricityFee + otherFees;
      
      return {
        monthly_rent: monthlyRent,
        electricity_fee: electricityFee,
        other_fees: otherFees,
        total_amount: totalAmount
      };
    } catch (error) {
      throw error;
    }
  }

  // 自動觸發月度收租
  async triggerMonthlyRent() {
    try {
      const activeContracts = await Contract.findAll({
        where: { status: 'active' }
      });

      const currentMonth = new Date().toISOString().slice(0, 7) + '-01';
      
      for (const contract of activeContracts) {
        const rentData = await this.calculateMonthlyRent(contract.id, currentMonth);
        
        await RentalRecord.create({
          contract_id: contract.id,
          billing_month: currentMonth,
          ...rentData,
          status: 'pending'
        });
      }
    } catch (error) {
      throw error;
    }
  }
}
```
